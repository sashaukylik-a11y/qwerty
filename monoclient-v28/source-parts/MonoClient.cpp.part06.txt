    if(g_gameWnd&&IsWindowVisible(g_gameWnd)){DWORD wp=0;GetWindowThreadProcessId(g_gameWnd,&wp);if(wp==g_target.pid){int n=GetWindowTextLengthW(g_gameWnd);if(n>0)GetWindowTextW(g_gameWnd,g_targetWindow,260);else g_targetWindow[0]=0;g_targetTitleMisses=0;return TRUE;}}

static DWORD WINAPI EngineWorker(LPVOID){
    SetThreadPriority(GetCurrentThread(),THREAD_PRIORITY_BELOW_NORMAL);
    ULONGLONG lastFastSearch=0,lastDeepSearch=0,lastIdentity=0,lastIdleAttach=0;
    while(!g_engineStop){
        ULONGLONG now=GetTickCount64();
        if(!g_target.pid){
            // Cheap path: top-level windows only. This catches normal Minecraft/Pulse/Java launches.
            if(now-lastFastSearch>=700){lastFastSearch=now;GatherCandidates(FALSE);if(g_target.pid)ResetBridgeAttempts();}
            // Full process snapshot is only a rare fallback for custom-title clients.
            if(!g_target.pid&&now-lastDeepSearch>=5000){lastDeepSearch=now;GatherCandidates(TRUE);if(g_target.pid)ResetBridgeAttempts();}
        }else if(now-lastIdentity>=1800){
            lastIdentity=now;
            if(!FastTargetAlive()){
                memset(&g_target,0,sizeof(g_target));g_targetHostMode=HOST_NONE;g_targetNeedsGameTitle=FALSE;g_targetTitleMisses=0;g_bridgePidHint=0;g_bridgeCreateHint=0;
                if(!g_bridgeAttachBusy)BridgeClose();ResetBridgeAttempts();lastFastSearch=0;lastDeepSearch=0;
            }
        }

        BOOL active=(g_target.pid&&GameForeground());

        // Connection state must not depend on the combat toggle. Previously this early-off path
        // skipped BridgeEnsure(), so the UI could find the Minecraft window but remain forever at
        // "JVM PID not selected / candidates 0" until TriggerBot itself was enabled.
        if(!g_cfg.triggerEnabled){
            if(g_target.pid&&now-lastIdleAttach>=1000){lastIdleAttach=now;BridgeEnsure();}
            Sleep(180);continue;
        }

        if(!active){
            // Keep attach asynchronous, but never poll Minecraft state while the game is not foreground.
            if(g_target.pid&&now-lastIdleAttach>=1000){lastIdleAttach=now;BridgeEnsure();}
            Sleep(100);continue;
        }

        // v28: fast target sampling for short flicks. BridgePoll exits early before heavy crit fields unless an attack is near-ready.
        BridgePoll();TriggerTick();Sleep(16);
    }
    return 0;
}

static BOOL ReadRightShiftDown(){
    if(GetAsyncKeyState(VK_RSHIFT)&0x8000)return TRUE;
    // Fallback for launchers/remote-input layers that expose only generic VK_SHIFT.
    // Exclude an actual left-shift press so LSHIFT never becomes the menu hotkey.
    return ((GetAsyncKeyState(VK_SHIFT)&0x8000)!=0)&&((GetAsyncKeyState(VK_LSHIFT)&0x8000)==0);
}
