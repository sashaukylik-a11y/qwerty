static U64 FindJvmBase(DWORD pid){

        if(HasHotSpotMarkersRange(exactJvm,g_bridge.jvmSpan)){g_bridge.heuristicStructs=1;return exactJvm;}

        // failure reason instead of incorrectly reporting that no JVM image exists.

    }
