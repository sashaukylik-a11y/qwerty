    // Keep the top-level window fixed at 900x600. Resizing a layered HWND every animation frame
    // forces DWM to rebuild its surface and was the main source of visible flashing on some GPUs.
    if(g_menuAnim>=999){BitBlt(target,0,0,900,600,g_menuDC,0,0,SRCCOPY);return;}
    FillRound(target,0,0,w,h,0,MENU_KEY);
    int scale=700+(300*g_menuAnim)/1000;int dw=900*scale/1000,dh=600*scale/1000;
    if(dw<1)dw=1;if(dh<1)dh=1;int dx=(900-dw)/2,dy=(600-dh)/2;
    SetStretchBltMode(target,COLORONCOLOR);StretchBlt(target,dx,dy,dw,dh,g_menuDC,0,0,900,600,SRCCOPY);
}
